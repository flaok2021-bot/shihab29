name: Build APK

on:
  push:
    branches: [ main ]
  workflow_dispatch:

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-node@v4
        with:
          node-version: 20

      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: 17

      - name: Install dependencies
        run: npm install

      - name: Add Android platform
        run: npx cap add android

      - name: Sync Capacitor
        run: npx cap sync android

      - name: Apply custom icons
        run: |
          unzip -o icons.zip -d icons_extracted
          cp -r icons_extracted/android_res/mipmap-* android/app/src/main/res/

      - name: Patch AndroidManifest
        run: node scripts/patch-manifest.js

      - name: Build APK
        run: |
          cd android
          chmod +x gradlew
          ./gradlew assembleDebug

      # ننسخ الـ APK لملف بالجذر باسم واضح — بدون هذا، GitHub يحافظ على
      # شجرة المجلدات كاملة (android/app/build/outputs/apk/debug/...) جوه
      # ملف الـ zip اللي تحمّله، فتحس إنك طلعت "ملفات" مو تطبيق واحد
      - name: Prepare APK for download
        run: cp android/app/build/outputs/apk/debug/app-debug.apk sqoor-app.apk

      - uses: actions/upload-artifact@v4
        with:
          name: sqoor-app
          path: sqoor-app.apk
